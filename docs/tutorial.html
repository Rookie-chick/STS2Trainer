二、原理：为什么"按方法名"改游戏有效

游戏启动
└─ .NET 运行时加载 sts2.dll（8.93 MB 托管程序集）
   └─ 元数据保存：命名空间 / 类 / 方法名 / 签名
      └─ 方法首次调用 → JIT 成机器码（得到代码地址）
         └─ 「方法描述符 ↔ 机器码地址」运行时一直持有
            └─ ClrMD / CE 的 Mono 接口能读出来  ⇒  名字→地址
路线做什么抗更新样本用了吗
① 改值直接改字段（如引擎时间缩放）★★用了（游戏速度）
② 打补丁（AA）方法入口 jmp 到自己的代码★★★★主力：17 define / 12 alloc
③ HarmonyPrefix/Postfix，不碰机器码★★★★★没用（但游戏自带 0Harmony.dll）

为什么"改这个方法"就有效

改什么为什么游戏会认
属性的 setter（set_CurrentHp / set_Block / set_Gold）所有改血/盾/钱的代码都走它 ⇒ 在入口拦一道，任何来源的改动都经过你
扣减方法（LoseEnergy / LoseStars）唯一减少入口 ⇒ 让它扣 0 就是无限
Hook 的返回值（ModifyDamage / ModifyMerchantPrice / ModifyMaxEnergy）游戏把数值交给 Hook 算 ⇒ 改返回值 = 改结算
概率 roll（PotionRewardOdds::Roll / CardRarityOdds::GetBaseOdds）掉落与稀有度都从这里出结果 ⇒ 改成必中
奖励上下限（get_MinGoldReward / get_MaxGoldReward）结算金币区间被改 ⇒ 金币倍率
反面教材（我在 Brotato 上真踩过）：直接每帧回写"金币余额" ⇒ 无效甚至闪退 ✗（余额由信号驱动、UI 重新拉值）。 结论：改"真值的产生处"，别改"显示出来的副本"。

// ============================================================================
//  sts2hp —— 杀戮尖塔 2「无限血」最小示例（零基础可照抄）
//
//  用法（在 PowerShell 里，先 cd 到本工程所在目录）：
//      dotnet run -- <游戏pid>            安装：血量锁 9999
//      dotnet run -- <游戏pid> 50000      安装：血量锁 50000
//      dotnet run -- <游戏pid> poke 1     验证：把血强行写成 1，看会不会被补回
//
//  它做三件事：
//      ① 找到"玩家对象"和它身上的"血量字段"
//      ② 找到"顶栏血量文字"的刷新方法（NTopBarHp.UpdateHealth）
//      ③ 找一个每帧都会执行的方法当"挂点"（NTargetManager._Process），
//         在它的入口打一个 5 字节跳转，跳到我们自己的代码里，
//         代码里：血不满 → 写满 → 调一次刷新方法 → 再跳回原方法
// ============================================================================

using System;
using System.Collections.Generic;
using System.Text;
using Microsoft.Diagnostics.Runtime;

class Prog
{
    // ---- 这两个类型名是从反编译源码里抄来的（decomp\full 下搜类名） ----
    const string T_PLAYER = "MegaCrit.Sts2.Core.Entities.Players.Player";
    const string T_CREATURE = "MegaCrit.Sts2.Core.Entities.Creatures.Creature";
    const string T_HPBAR = "MegaCrit.sts2.Core.Nodes.TopBar.NTopBarHp";   // 注意命名空间是小写 sts2
    const string T_TARGETMGR = "MegaCrit.Sts2.Core.Nodes.Combat.NTargetManager";

    // ---- 字段真实偏移 = ClrMD 报的 offset + 8（为什么 +8 见教程第 7.3 节） ----
    const int OFF_PLAYER_CREATURE = 0x68;  // Player.Creature
    const int OFF_CUR_HP = 0x84;           // Creature._currentHp
    const int OFF_MAX_HP = 0x88;           // Creature._maxHp
    const int OFF_BAR_PLAYER = 0x58;       // NTopBarHp._player

    static DataTarget dt;
    static ClrRuntime rt;
    static Dictionary<string, List<ClrObject>> heap;

    static void Main(string[] args)
    {
        Console.OutputEncoding = Encoding.UTF8;
        if (args.Length < 1)
        {
            Console.WriteLine("用法: dotnet run -- <游戏pid> [血量]");
            Console.WriteLine("      dotnet run -- <游戏pid> poke <值>");
            return;
        }
        int pid = int.Parse(args[0]);

        Console.WriteLine("【1】连接游戏进程 pid=" + pid);
        dt = DataTarget.AttachToProcess(pid, suspend: false);
        rt = dt.ClrVersions[0].CreateRuntime();
        Console.WriteLine("     CLR 版本 = " + dt.ClrVersions[0].Version + "（能打印出来就说明连上了）");

        Console.WriteLine("【2】扫描托管堆（第一次会花几秒）");
        BuildHeap();
        Console.WriteLine("     堆里共 " + heap.Count + " 种类型");

        if (args.Length > 2 && args[1] == "poke")
        {
            Poke(pid, int.Parse(args[2]));
            return;
        }
        int hpValue = args.Length > 1 ? int.Parse(args[1]) : 9999;
        Install(pid, hpValue);
    }

    // ---------------------------------------------------------------- 堆缓存
    static void BuildHeap()
    {
        heap = new Dictionary<string, List<ClrObject>>();
        foreach (var obj in rt.Heap.EnumerateObjects())
        {
            var t = obj.Type;
            if (t == null || t.Name == null) continue;
            if (!heap.TryGetValue(t.Name, out var list)) { list = new List<ClrObject>(); heap[t.Name] = list; }
            list.Add(obj);
        }
    }

    static List<ClrObject> Of(string typeName)
    {
        return heap.TryGetValue(typeName, out var l) ? l : new List<ClrObject>();
    }

    // ---------------------------------------------------------------- 读内存小工具
    static long ReadI64(ulong addr)
    {
        var buf = new byte[8];
        if (dt.DataReader.Read(addr, buf) != 8) return 0;
        return BitConverter.ToInt64(buf, 0);
    }

    static int ReadI32(ulong addr)
    {
        var buf = new byte[4];
        if (dt.DataReader.Read(addr, buf) != 4) return -1;
        return BitConverter.ToInt32(buf, 0);
    }

    // ---------------------------------------------------------------- 找玩家
    static ulong FindPlayerAndCreature(out ulong creature)
    {
        creature = 0;
        int n = 0;
        foreach (var obj in Of(T_PLAYER))
        {
            n++;
            ulong p = obj.Address;
            ulong c = (ulong)ReadI64(p + OFF_PLAYER_CREATURE);
            Console.WriteLine("     Player#" + n + " 地址=0x" + p.ToString("X") + "  Creature=0x" + c.ToString("X"));
            if (c > 0x10000 && creature == 0) creature = c;
        }
        if (creature == 0) return 0;
        foreach (var obj in Of(T_PLAYER)) return obj.Address;
        return 0;
    }

    static ulong FindBar(ulong player)
    {
        foreach (var obj in Of(T_HPBAR))
        {
            var t = obj.Type;
            if (t == null || t.Name != T_HPBAR) continue;
            ulong pl = (ulong)ReadI64(obj.Address + OFF_BAR_PLAYER);
            Console.WriteLine("     NTopBarHp 实例=0x" + obj.Address.ToString("X") + "  _player=0x" + pl.ToString("X"));
            if (pl == player) return obj.Address;
        }
        return 0;
    }

    static ulong MethodAddr(string typeName, string methodName)
    {
        if (!heap.TryGetValue(typeName, out var list) || list.Count == 0)
        {
            Console.WriteLine("     ✗ 堆里找不到类型 " + typeName);
            return 0;
        }
        foreach (var m in list[0].Type.Methods)
        {
            if (m.Name != methodName) continue;
            Console.WriteLine("     " + methodName + " native=0x" + m.NativeCode.ToString("X"));
            if (m.NativeCode > 0x10000 && m.NativeCode != ulong.MaxValue) return (ulong)m.NativeCode;
            Console.WriteLine("       （0xFFFFFFFFFFFFFFFF = 这个方法还没被 JIT，换个方法或等它被调用过）");
        }
        return 0;
    }

    // ---------------------------------------------------------------- 安装
    static void Install(int pid, int hpValue)
    {
        Console.WriteLine("【3】找玩家对象");
        ulong player;
        ulong creature;
        ulong p = FindPlayerAndCreature(out creature);
        if (creature == 0) { Console.WriteLine("     ✗ 没找到 Player/Creature（是不是还没进战斗？）"); return; }
        player = p;
        Console.WriteLine("     ✓ 玩家 Creature = 0x" + creature.ToString("X"));
        Console.WriteLine("     当前血量 = " + ReadI32(creature + OFF_CUR_HP) + "/" + ReadI32(creature + OFF_MAX_HP)
                          + "（应该和画面上的数字一致，不一致就是偏移算错了）");

        Console.WriteLine("【4】找顶栏血量节点 + 它的刷新方法");
        ulong bar = FindBar(player);
        if (bar == 0) { Console.WriteLine("     ✗ 没找到 NTopBarHp 实例"); return; }
        ulong updateHealth = MethodAddr(T_HPBAR, "UpdateHealth");
        if (updateHealth == 0) { Console.WriteLine("     ✗ 没拿到刷新方法地址"); return; }

        Console.WriteLine("【5】找每帧执行的挂点 NTargetManager._Process");
        ulong proc = MethodAddr(T_TARGETMGR, "_Process");
        if (proc == 0)
        {
            Console.WriteLine("     ✗ 没拿到挂点（这个类型要在战斗里才有实例；也可手动传地址）");
            return;
        }

        Console.WriteLine("【6】先把血量上限改成 " + hpValue);
        var h = Hook.Open(pid);
        Hook.WriteI32(h, creature + OFF_MAX_HP, hpValue);
        Console.WriteLine("     回读 = " + Hook.ReadI32(h, creature + OFF_CUR_HP) + "/" + Hook.ReadI32(h, creature + OFF_MAX_HP));

        Console.WriteLine("【7】在挂点上打补丁（跳转到我们的代码）");
        Hook.InstallFrameHook(pid, proc, creature + OFF_CUR_HP, hpValue, bar, updateHealth);

        Console.WriteLine("【8】完成。现在看游戏：顶栏应该是 " + hpValue + "/" + hpValue);
        Console.WriteLine("     验证：dotnet run -- " + pid + " poke 1");
    }

    // ---------------------------------------------------------------- 验证
    static void Poke(int pid, int v)
    {
        ulong creature;
        FindPlayerAndCreature(out creature);
        if (creature == 0) { Console.WriteLine("✗ 没找到 Creature"); return; }
        var h = Hook.Open(pid);
        ulong addr = creature + OFF_CUR_HP;
        Console.WriteLine("写入前 血=" + Hook.ReadI32(h, addr));
        Hook.WriteI32(h, addr, v);
        Console.WriteLine("已强行写成 " + v + "，立刻回读=" + Hook.ReadI32(h, addr));
        System.Threading.Thread.Sleep(500);
        Console.WriteLine("500ms 后回读 血=" + Hook.ReadI32(h, addr) + "  ← 被我们的补丁补回 = 成功");
    }
}

5.3 Hook.cs（注入器，逐行都有注释）

// ============================================================================
//  Hook.cs —— 往别人的进程里写机器码（inline hook）
//
//  原理（一张图）：
//
//    游戏原来的方法:                   打完补丁之后:
//    NTargetManager._Process            NTargetManager._Process
//      53                push rbx        E9 xx xx xx xx   jmp 我们的代码洞
//      48 83 EC 20       sub rsp,20      （剩下的字节被 nop 掉）
//      ... 方法正文 ...                   ...
//
//    我们分配的"代码洞"（在挂点附近 2GB 以内，保证 jmp 跳得到）:
//      push rcx / rdx / r8 / r9          ; ★保存现场（被挂钩方法的参数）
//      sub rsp,38h
//      mov rax, 血量字段地址
//      mov edx, [rax]                    ; 读出当前血量
//      cmp edx, 期望值
//      je 跳过
//      mov dword [rax], 期望值            ; ← 把血写满
//      movups [rsp+20h], xmm0            ; 保存浮点参数（_Process 有 double delta）
//      mov rcx, 顶栏节点
//      mov edx, 0 / xor r8d, r8d
//      mov rax, UpdateHealth             ; ← 调游戏的刷新方法
//      call rax
//      movups xmm0, [rsp+20h]
//    跳过:
//      add rsp,38h / pop r9/r8/rdx/rcx    ; ★还原现场（漏了这步游戏会崩！）
//      push rbx                          ; 把被我们覆盖的原指令补回来
//      sub rsp,20
//      jmp 回 挂点+5
// ============================================================================

using System;
using System.Collections.Generic;
using System.Runtime.InteropServices;

static class Hook
{
    const int PAGE_EXECUTE_READWRITE = 0x40;
    const int MEM_COMMIT = 0x1000, MEM_RESERVE = 0x2000;

    [DllImport("kernel32.dll", SetLastError = true)] static extern IntPtr OpenProcess(int access, bool inherit, int pid);
    [DllImport("kernel32.dll", SetLastError = true)] static extern bool ReadProcessMemory(IntPtr h, IntPtr addr, byte[] buf, IntPtr size, out IntPtr read);
    [DllImport("kernel32.dll", SetLastError = true)] static extern bool WriteProcessMemory(IntPtr h, IntPtr addr, byte[] buf, IntPtr size, out IntPtr written);
    [DllImport("kernel32.dll", SetLastError = true)] static extern IntPtr VirtualAllocEx(IntPtr h, IntPtr addr, IntPtr size, int type, int prot);
    [DllImport("kernel32.dll", SetLastError = true)] static extern bool VirtualProtectEx(IntPtr h, IntPtr addr, IntPtr size, int newProt, out int oldProt);
    [DllImport("kernel32.dll", SetLastError = true)] static extern bool FlushInstructionCache(IntPtr h, IntPtr addr, IntPtr size);

    public static IntPtr Open(int pid) { return OpenProcess(0x1F0FFF, false, pid); }

    public static byte[] Read(IntPtr h, ulong addr, int n)
    {
        var b = new byte[n];
        IntPtr got;
        if (!ReadProcessMemory(h, (IntPtr)(long)addr, b, (IntPtr)n, out got)) return null;
        return b;
    }

    public static int ReadI32(IntPtr h, ulong addr)
    {
        var b = Read(h, addr, 4);
        return b == null ? -1 : BitConverter.ToInt32(b, 0);
    }

    public static void WriteI32(IntPtr h, ulong addr, int value)
    {
        IntPtr w;
        WriteProcessMemory(h, (IntPtr)(long)addr, BitConverter.GetBytes(value), (IntPtr)4, out w);
    }

    public static string Hex(byte[] b)
    {
        if (b == null) return "<读不到>";
        var sb = new System.Text.StringBuilder();
        foreach (var x in b) sb.Append(x.ToString("X2")).Append(' ');
        return sb.ToString().Trim();
    }

    // 在目标地址附近（±2GB 内，保证 E9 相对跳转够得着）分配可执行内存
    static ulong AllocNear(IntPtr h, ulong target, int size)
    {
        long t = (long)target;
        for (long d = 0; d < 0x7000_0000L; d += 0x10000)
        {
            for (int s = 0; s < 2; s++)
            {
                long cand = s == 0 ? t - d : t + d;
                if (cand < 0x10000) continue;
                IntPtr p = VirtualAllocEx(h, (IntPtr)cand, (IntPtr)size, MEM_RESERVE | MEM_COMMIT, PAGE_EXECUTE_READWRITE);
                if (p != IntPtr.Zero) return (ulong)p;
            }
        }
        return 0;
    }

    // 一条 x86 指令有多长？只认 .NET 方法序言里常见的那几种（够用）
    static int InsnLen(byte[] b, int i)
    {
        byte op = b[i];
        if (op >= 0x50 && op <= 0x57) return 1;                                   // push/pop r64
        if (op == 0x41 && b[i + 1] >= 0x50 && b[i + 1] <= 0x57) return 2;         // push/pop r8-r15
        if (op == 0x48 && b[i + 1] == 0x83 && (b[i + 2] == 0xEC || b[i + 2] == 0xC4)) return 4;  // sub/add rsp,imm8
        if (op == 0x48 && b[i + 1] == 0x81 && (b[i + 2] == 0xEC || b[i + 2] == 0xC4)) return 7;  // sub/add rsp,imm32
        if (op == 0x48 && b[i + 1] >= 0xB8 && b[i + 1] <= 0xBF) return 10;         // mov r64,imm64
        if (op == 0x4C && b[i + 1] == 0x8B && b[i + 2] == 0xDC) return 3;          // mov r11,rsp
        return -1;
    }

    static int PrologueLen(byte[] head)
    {
        int n = 0;
        while (n < 5)
        {
            int len = InsnLen(head, n);
            if (len <= 0) return -1;
            n += len;
            if (n > 15) return -1;
        }
        return n;
    }

    // ---------------------------------------------------------------- 主入口
    public static void InstallFrameHook(int pid, ulong patchAddr, ulong fieldAddr, int value, ulong callThis, ulong callFn)
    {
        var h = Open(pid);
        if (h == IntPtr.Zero) { Console.WriteLine("     ✗ OpenProcess 失败（用管理员运行 PowerShell）"); return; }

        // 安全闸：地址不像话就别写，否则游戏会崩
        if (patchAddr < 0x10000 || patchAddr == ulong.MaxValue ||
            fieldAddr < 0x10000 || callThis < 0x10000 || callFn < 0x10000 || callFn == ulong.MaxValue)
        {
            Console.WriteLine("     ✗ 地址不合法，已中止（不写任何补丁）"); return;
        }

        var head = Read(h, patchAddr, 16);
        int n = PrologueLen(head);
        if (n < 5)
        {
            Console.WriteLine("     ✗ 读不到能整段覆盖的序言: " + Hex(head)); return;
        }
        var original = new byte[n];
        Array.Copy(head, 0, original, 0, n);
        Console.WriteLine("     挂点前 " + n + " 字节 = " + Hex(original));

        ulong cave = AllocNear(h, patchAddr, 0x1000);
        if (cave == 0) { Console.WriteLine("     ✗ 分配代码洞失败"); return; }

        // ---------------- 拼机器码 ----------------
        var code = new List<byte>();
        void B(params byte[] bs) { code.AddRange(bs); }
        void I32(int v) { code.AddRange(BitConverter.GetBytes(v)); }
        void I64(ulong v) { code.AddRange(BitConverter.GetBytes(v)); }

        // ★★★ 必须先保存现场：rcx/rdx/r8/r9 是"被挂钩方法自己的参数"，
        //     我们借用了它们去调刷新方法，用完必须原样还回去，否则游戏会崩
        B(0x51);                            // push rcx   （this）
        B(0x52);                            // push rdx
        B(0x41, 0x50);                      // push r8
        B(0x41, 0x51);                      // push r9
        B(0x48, 0x83, 0xEC, 0x38);          // sub rsp,38h（0x20 影子空间 + 0x10 存 xmm0 + 对齐）

        B(0x48, 0xB8); I64(fieldAddr);      // mov rax, &_currentHp
        B(0x8B, 0x10);                      // mov edx,[rax]
        B(0x81, 0xFA); I32(value);          // cmp edx, value
        int jeAt = code.Count; B(0x74, 0x00);  // je skip（位移待会补）
        B(0xC7, 0x00); I32(value);          // mov dword [rax], value
        B(0x0F, 0x11, 0x44, 0x24, 0x20);    // movups [rsp+20h], xmm0（_Process 的 double 参数）
        B(0x48, 0xB9); I64(callThis);       // mov rcx, 顶栏节点
        B(0x31, 0xD2);                      // xor edx,edx
        B(0x45, 0x31, 0xC0);                // xor r8d,r8d
        B(0x48, 0xB8); I64(callFn);         // mov rax, UpdateHealth
        B(0xFF, 0xD0);                      // call rax
        B(0x0F, 0x10, 0x44, 0x24, 0x20);    // movups xmm0,[rsp+20h]
        int skipAt = code.Count;
        code[jeAt + 1] = (byte)(skipAt - (jeAt + 2));   // 补 je 的位移

        // ★ 还原现场
        B(0x48, 0x83, 0xC4, 0x38);          // add rsp,38h
        B(0x41, 0x59);                      // pop r9
        B(0x41, 0x58);                      // pop r8
        B(0x5A);                            // pop rdx
        B(0x59);                            // pop rcx

        foreach (var b in original) B(b);    // 补回被覆盖的原指令

        int jmpAt = code.Count;
        B(0xE9); I32(0);                     // jmp 回 patchAddr+n
        int back = (int)((long)(patchAddr + (ulong)n) - (long)(cave + (ulong)jmpAt + 5));
        var rb = BitConverter.GetBytes(back);
        code[jmpAt + 1] = rb[0]; code[jmpAt + 2] = rb[1]; code[jmpAt + 3] = rb[2]; code[jmpAt + 4] = rb[3];

        IntPtr w;
        if (!WriteProcessMemory(h, (IntPtr)(long)cave, code.ToArray(), (IntPtr)code.Count, out w))
        { Console.WriteLine("     ✗ 写代码洞失败"); return; }

        // ---------------- 打补丁 ----------------
        var patch = new byte[n];
        patch[0] = 0xE9;
        var d = BitConverter.GetBytes((int)((long)cave - (long)(patchAddr + 5)));
        patch[1] = d[0]; patch[2] = d[1]; patch[3] = d[2]; patch[4] = d[3];
        for (int i = 5; i < n; i++) patch[i] = 0x90;      // 多出来的补 nop

        int oldProt;
        VirtualProtectEx(h, (IntPtr)(long)patchAddr, (IntPtr)n, PAGE_EXECUTE_READWRITE, out oldProt);
        WriteProcessMemory(h, (IntPtr)(long)patchAddr, patch, (IntPtr)n, out w);
        int tmp;
        VirtualProtectEx(h, (IntPtr)(long)patchAddr, (IntPtr)n, oldProt, out tmp);
        FlushInstructionCache(h, (IntPtr)(long)patchAddr, (IntPtr)n);   // 告诉 CPU 指令变了

        Console.WriteLine("     代码洞 0x" + cave.ToString("X") + "（" + code.Count + " 字节）");
        Console.WriteLine("     代码洞内容: " + Hex(code.ToArray()));
        Console.WriteLine("     补丁: " + Hex(patch) + "   回读: " + Hex(Read(h, patchAddr, n)));
        Console.WriteLine("     ✓ 补丁已生效");
    }
}

10. 代码怎么实现（Hook.cs 逐段讲解）

10.1 四个 Windows API（全部只要名字看懂）

API一句话作用
OpenProcess拿到操作游戏进程的"句柄"（要管理员）
VirtualAllocEx在游戏进程里申请一块内存（放我们的机器码）
WriteProcessMemory往游戏内存写字节
VirtualProtectEx改内存保护属性（代码段默认只读，要改成可写才能打补丁）
FlushInstructionCache告诉 CPU"这段指令缓存作废了，重新读"（不改这个，补丁可能不生效）

10.2 为什么代码洞要"分配在挂点附近"

E9 后面跟的是相对位移（4 字节有符号），最多跳 ±2GB。

如果代码洞被分配到很远的地方，jmp 会算错 → 崩。

所以 AllocNear() 从挂点开始，以 64KB 为步长向两边试，第一个成功的就用。

10.3 为什么要"补回原指令"

我们只借用了方法开头 5 个字节。执行完自己的逻辑后，方法还得继续跑它原来的代码，

所以必须把那 5 个字节（可能是 push rbx; sub rsp,20h）原样执行一遍，再 jmp 回"第 5 字节之后"。

10.4 ★ 为什么必须保存 rcx/rdx/r8/r9 和 xmm0（我在这里崩过一次游戏）

NTargetManager._Process 的 this 在 rcx 里，方法正文还要用它。

我们的代码洞里为了调 UpdateHealth，把 rcx 改成了"顶栏节点"——

如果不还回去，方法正文就会拿错 this，直接崩。

我第一次写精简版时就漏了这步，测试时游戏当场退出。加上 push/pop 之后一切正常。

同理 xmm0 里放着 _Process 的 double delta 参数，我们调用别的托管方法会覆盖它，

所以也要 movups 存到栈上再读回。

10.5 栈对齐（为什么 sub rsp,38h）

Windows x64 约定：call 之前 rsp 必须是 16 的倍数，且要给被调函数留 0x20 字节"影子空间"。

我们的代码：

  • 进代码洞时 rsp % 16 == 8（因为是被 call/jmp 进来的）
  • push 4 个寄存器 → 还是 8
  • sub rsp,0x38（56）→ 变 0（对齐了，其中 0x20 影子空间 + 0x10 存 xmm0 + 8 填充）
  • 于是 call rax 进去时被调函数看到 rsp % 16 == 8，符合约定 ✓
不对齐会怎样
被调方法里一旦用到 SSE 对齐指令（vmovdqa）就会崩，而且崩得很随机，很难查。

10.6 安全闸（为什么先检查地址）

if (patchAddr < 0x10000 || patchAddr == ulong.MaxValue || ...) { 中止 }

ClrMD 用 0xFFFFFFFFFFFFFFFF 表示"这个方法没有机器码（没 JIT）"。

我第一版没判断，把这个值当成地址去 call，游戏立刻崩溃。

现在任何"要 call 的地址"都会先过一遍这种检查，不合法就一个字节都不写。

{

    private set
    {

2.2 为什么会这样

我们的代码洞 = 手写的机器码，没有 .NET 的"异常展开信息(unwind info)"
游戏方法内部抛出 NullReferenceException（_hsv 为 null）
  → CLR 要往上找谁来 catch
  → 展开到我们这段裸机器码时找不到帧信息
  → 直接 FailFast / 访问违例：进程被终止

2.3 解法：把"调用目标"换成最轻的那个

候选做了什么安全性
OnStarsChanged(old,new)动画 + 刷文字 + 刷显隐（碰 _hsv）✗ 会抛 NRE
SetStarCountText(int)刷文字 + 碰 _hsv✗ 同样危险
RefreshVisibility()只碰 Visible✅ 本项目最终用它
不调（只写字段）——✅ 但星星图标可能一直不显示（显隐没刷新过）

数字本来由 _Process 每帧刷，所以我们只需要"让它显形" ⇒ 调 RefreshVisibility 就够。

这一篇教第三种钩子：强制返回
前两个功能用了"帧钩子（轮询写字段）"和"setter 钩子（改参数）"。
奖励类功能大多是 bool / int / enum 的"算一下返回个决定"——

2. 原理：药水是怎么"掷骰子"的

3. 我是怎么做的：怎么找出这个方法的（真实路径）

① ILSpy 里 Ctrl+F 搜 "potion"（选 All）
   → 命中 MegaCrit.Sts2.Core.Odds.PotionRewardOdds
② 类里只有一个"掷骰子"方法：public bool Roll(Player, AscensionManager, RoomType)
   → 返回 bool ⇒ 典型"强制返回"目标
③ 搜 "Roll(" 的调用点 → 确认战斗结算时用它决定奖励里放不放药水
④ ClrMD 看有没有机器码
dotnet run --project D:\sts2\clr\sts2clr -- <pid> methods MegaCrit.Sts2.Core.Odds.PotionRewardOdds Roll

两种结果都要会处理：

A. native=0x7FF7xxxxxxxx              → 直接挂
B. native=0xFFFFFFFFFFFFFFFF          → 还没 JIT（要等第一次算奖励）
   ⇒ 延迟安装：主循环每 2 秒重查一次，一拿到地址立刻挂上（第一次奖励赶不上，第二次起生效）
为什么会是 B
.NET 是"用到才编译"。Roll 只有"战斗结束算奖励"那一刻才第一次被调用，
战斗中途装它时它可能压根还没有机器码。这就是"延迟安装"存在的原因。

4.1 钩子本体

public static byte[] InstallForceReturnHook(IntPtr h, ulong methodAddr, ulong flagAddr, int retValue)
{
    var head = Read(h, methodAddr, 16);
    int n = PrologueLen(head);                        // 覆盖 ≥5 字节的整指令长度
    var original = new byte[n]; Array.Copy(head, 0, original, 0, n);
    ulong cave = AllocNear(h, methodAddr, 0x1000);    // E9 只能跳 ±2GB

    var c = new List<byte>();
    B(0x50);                                  // push rax
    B(0x48, 0xB8); I64(flagAddr);             // mov rax,&开关
    B(0x83, 0x38, 0x00);                      // cmp dword [rax],0
    int jeAt = c.Count; B(0x74, 0x00);        // je  原样
    B(0x58);                                  // pop rax
    B(0xB8); I32(retValue);                   // mov eax,返回值
    B(0xC3);                                  // ret        ← 直接返回
    int origAt = c.Count;
    c[jeAt + 1] = (byte)(origAt - (jeAt + 2));
    foreach (var b in original) B(b);         // 原样路径
    B(0xE9); I32(back);                       // jmp 回 方法+n

    WriteProcessMemory(h, cave, code);
    Patch(h, methodAddr, cave, n);            // 入口改 E9 + nop…
    return original;                          // 退出时还原用
}

生成的机器码：

50                 push rax
48 B8 <&flag>      mov  rax,&开关
83 38 00           cmp  dword [rax],0
74 07              je   原样
58                 pop  rax
B8 01 00 00 00     mov  eax,1        ; true
C3                 ret
原样: <原 n 字节> ; E9 跳回

4. 你该怎么做：代码怎么写

4.1 decimal 的内存布局（记住这个，一辈子用得上）

decimal = 16 字节 = 4 个 int
  +0x00 lo     低 32 位
  +0x04 mid    中 32 位
  +0x08 hi     高 32 位
  +0x0C flags  bit16~23 = scale（小数点位置），bit31 = 符号

  1     = lo=1,    mid=0, hi=0, flags=0x00000000
  1.0   = lo=10,   mid=0, hi=0, flags=0x00010000 (scale=1)
  0     = lo=0,    mid=0, hi=0, flags=0
  9999  = lo=9999, mid=0, hi=0, flags=0
想改成 50%？0.5m = [5, 0, 0, 0x00010000]。别手算，用 C# 的 decimal.GetBits(0.5m)。

4.2 钩子：把返回值写成 1

public static byte[] InstallForceDecimalReturnHook(IntPtr h, ulong methodAddr, ulong flagAddr, decimal value)
{
    int[] parts = decimal.GetBits(value);        // [lo, mid, hi, flags]
    var head = Read(h, methodAddr, 16);
    int n = PrologueLen(head);
    var original = new byte[n]; Array.Copy(head, 0, original, 0, n);
    ulong cave = AllocNear(h, methodAddr, 0x1000);

    var c = new List<byte>();
    B(0x50);                                     // push rax
    B(0x48, 0xB8); I64(flagAddr);                // mov rax,&开关
    B(0x83, 0x38, 0x00);                         // cmp dword [rax],0
    int jeAt = c.Count; B(0x74, 0x00);           // je 原样
    B(0x58);                                     // pop rax
    B(0xC7, 0x01); I32(parts[0]);                // mov dword [rcx],   lo
    B(0xC7, 0x41, 0x04); I32(parts[1]);          // mov dword [rcx+4], mid
    B(0xC7, 0x41, 0x08); I32(parts[2]);          // mov dword [rcx+8], hi
    B(0xC7, 0x41, 0x0C); I32(parts[3]);          // mov dword [rcx+0xC], flags
    B(0x48, 0x89, 0xC8);                         // mov rax,rcx   ← ★必须
    B(0xC3);                                     // ret
    // …原样路径 + jmp 回 方法+n（同 14 篇）…
    return original;
}

// MegaCrit.Sts2.Core.Hooks/Hook.cs
public static bool ShouldTakeExtraTurn(ICombatState combatState, Player player)
{
    foreach (AbstractModel item in IterateCombatHookListeners(combatState))
        if (item.ShouldTakeExtraTurn(player)) return true;    // ← 任何一个模型说要，就返回 true
    return false;
}
  • 只要任何一个遗物/能力/怪物模型说"要额外回合"，它就返回 true。
  • 我们让它永远返回 false ⇒ 谁都不能要额外行动。
这个功能"作用对象"要看清
名字里带 player，但它是被"敌人回合调度"调用的（问"这个玩家这一侧要不要给额外行动"）。

{
    bool flag = CurrentHp > 0 && amount >= (decimal)CurrentHp;

    ...

  • 它每个生物（玩家/敌人）受伤都会走，参数 amount 是"这一下要扣多少血"。

4.2 decimal 参数的内存布局（和 15 篇一样）

rdx ──► +0x00 lo(int)   低 32 位
        +0x04 mid(int)
        +0x08 hi(int)
        +0x0C flags(int)  scale 在 bit16~23
三条命门
1. 必须 push/pop 所有被改的寄存器（尤其 rcx：我们用它当门禁比较对象）
2. 必须在指针上写满 16 字节（只写 lo，mid/hi/flags 残留会变成天文数字）
3. 必须门禁：target == 玩家 时放行，否则自己也被秒
// Hook.cs：改 decimal 参数的钩子
public static byte[] InstallDecimalArgHook(IntPtr h, ulong methodAddr, ulong flagAddr,
                                           ulong playerCreature, int newLo)
{
    var head = Read(h, methodAddr, 16);
    int n = PrologueLen(head);
    var original = new byte[n]; Array.Copy(head, 0, original, 0, n);
    ulong cave = AllocNear(h, methodAddr, 0x1000);

    var c = new List<byte>();
    void B(params byte[] bs) => c.AddRange(bs);
    void I32(int v) => c.AddRange(BitConverter.GetBytes(v));
    void I64(ulong v) => c.AddRange(BitConverter.GetBytes(v));

    B(0x50);                                  // push rax
    B(0x48, 0xB8); I64(flagAddr);             // mov rax,&开关
    B(0x83, 0x38, 0x00);                      // cmp dword [rax],0
    int jeFlag = c.Count; B(0x74, 0x00);      // je 原样（功能关）

    B(0x48, 0xB8); I64(playerCreature);       // mov rax,玩家Creature
    B(0x48, 0x39, 0xC1);                      // cmp rcx,rax       ← rcx = 受伤者
    int jePlayer = c.Count; B(0x74, 0x00);    // je 原样（玩家自己别改！）

    B(0xC7, 0x02); I32(newLo);                // mov dword [rdx],9999
    B(0xC7, 0x42, 0x04); I32(0);              // mov dword [rdx+4],0
    B(0xC7, 0x42, 0x08); I32(0);              // mov dword [rdx+8],0
    B(0xC7, 0x42, 0x0C); I32(0);              // mov dword [rdx+0xC],0

    int origAt = c.Count;
    c[jeFlag + 1] = (byte)(origAt - (jeFlag + 2));   // 两条 je 都跳到"原样"
    c[jePlayer + 1] = (byte)(origAt - (jePlayer + 2));

    B(0x58);                                  // pop rax
    foreach (var b in original) B(b);
    int jmpAt = c.Count; B(0xE9); I32(0);
    // …填 rel32 / 写代码洞 / Patch(入口)…
    return original;
}
三条命门
1. 必须 push rax / pop rax：我们改了 rax，方法体可能还要用。
2. 必须在 [rdx] 上写 16 字节：只写 lo 不够，mid/hi/flags 残留会让数值变成天文数字。
3. 必须门禁：cmp rcx, 玩家Creature —— 否则玩家的受伤也会被改成 9999（自己秒自己）。

19 篇是"把数值改成固定值"，这一篇是"把数值乘 N"
技术上多一步：读 decimal → 乘 → 写回。
只要数值是整数（mid=hi=0、scale=0），乘 lo 就够了——这是 90% 的伤害/金币场景。
真要处理小数（比如 ×1.5），就得调 System.Decimal.op_Multiply（本篇章末讲思路）。
热键：Num .（小键盘小数点）

1. 目标

项内容

2. 原理

// 19 篇的同一层：Creature.LoseHpInternal(decimal amount, ValueProp props)
// rcx = this(受伤者)   rdx = &amount(decimal)   r8d = props

差别只在代码洞里做什么：

篇代码洞里干的事
19（超级伤害）[rdx] = 9999（覆盖）
20（倍率）eax = [rdx] * N; [rdx] = eax（读-乘-写）

2.1 为什么可以先只乘 lo

decimal 的"整数值"一定满足：mid = 0, hi = 0, flags(scale) = 0
  6     → lo=6,    mid=0, hi=0, flags=0
  9999  → lo=9999, mid=0, hi=0, flags=0
⇒ 乘一个整数 N，结果仍然只需要 lo 部分（只要不溢出 2^31）
如果伤害可能是小数（例如 2.5），lo 里存的是 25，scale=1。
这时直接 lo*10 得到 250 —— 而 scale 还是 1 ⇒ 实际值 25.0，等于没变。
这种情况下要么把 scale 也改掉，要么走"调 op_Multiply"的正规路（见第 6 节）。

二十五、功能篇 14/20：金币倍率

上接：21-复刻-防御倍率 · 下接：23-复刻-商店免费

这一篇补最后一个"返回值类型"：int（eax）
到这儿四种返回值全齐了：bool/enum/int → eax、float → xmm0、decimal → rcx 隐藏指针。

项内容

2. 原理

// MegaCrit.Sts2.Core.Models/EncounterModel.cs

public virtual int MaxGoldReward { get; }      // ← int
// 战斗结算时：gold = rng.Next(encounter.MinGoldReward, encounter.MaxGoldReward)

改这两个 get 的返回值 = 直接改"这场给多少钱"。

改 _gold 是 10-复刻-金币无限（锁值）；这个功能是"奖励变多"，

三十、功能篇 19/20：分数倍率

上接：26-复刻-最高分数 · 下接：28-复刻-游戏速度

这是"同一个方法、两种玩法"的典型：26 篇写死大数，这一篇乘 N
难点只有一个：要拿到原返回值再乘 ⇒ 不能在入口 ret，要在方法出口改 eax。
本篇讲"尾部 hook（trampoline at return）"的最小可行做法。

1. 目标

项内容

2. 原理：为什么不能照抄 26 篇

26 篇（写死）：入口 mov eax,999999; ret        → 直接扔掉原逻辑
27 篇（倍率）：要 eax = 原返回值 × 10           → 必须等原方法算完

方法算完的那一刻 = 执行 ret 的那一刻。所以要在 每个 ret 之前插一段"eax *= 10"。

2.1 找 ret 的两种办法

办法做法难度
A. 扫指令从入口往后扫，找到所有 C3(ret) / C2 imm16(ret n)，把每个 ret 前 5 字节改成 jmp 代码洞（代码洞里 imul eax,eax,10; <原 ret>）；要自己写一个能正确切分指令的小解析器中高（指令长度要算对）
B. 换目标不 hook CalculateScore，改 hook 更小、只算一部分 的方法（例如 ScoreUtility 里各分项相加的地方）中（要读懂分数构成）
C. 合并到 26 篇放弃"倍率"，直接用"写死大数"低（推荐新手）
给零基础的建议
先做 C（26 篇的写死版），它已经把"分数很高"这个目的达到了。
想做"原分 ×10"再上 A/B——这是本套教程里第一个"需要自己写指令扫描器"的功能，属于进阶。

3. 我是怎么做的：怎么找出这个方法的（真实路径）

③ 决定策略：先做 C（合并到 26 篇），A/B 作为进阶留档

// 1) 从入口往后扫，收集所有整指令边界上的 C3 / C2
List<ulong> FindRets(IntPtr h, ulong entry, int maxLen = 400)
{
    var rets = new List<ulong>();
    ulong p = entry;
    for (int i = 0; i < maxLen; i++)
    {
        byte[] b = Read(h, p, 16);
        int len = InsnLen(b, 0);                 // 08 篇的指令长度函数
        if (len <= 0) break;                     // 认不出就停（别硬扫）
        if (b[0] == 0xC3) rets.Add(p);           // ret
        if (b[0] == 0xC2 && len == 3) rets.Add(p); // ret imm16
        p += (ulong)len;
    }
    return rets;
}

// 2) 每个 ret 前插一段：jmp 代码洞 → imul eax,eax,10 → 原 ret → 回原位置
//    注意：ret 只有 1 字节，我们至少要覆盖 5 字节
//    ⇒ 只能改"ret + 后面的填充字节(CC/90)"，或者把 ret 之前的指令一起挪走
这里有个硬约束
jmp rel32 要 5 字节，而 ret 只有 1 字节。
所以"在 ret 前插 jmp"必须吃掉 ret 前面至少 4 字节的完整指令，把它们一起搬到代码洞里。
这就是为什么方案 A 需要"能正确切分指令"——没有它就会把一条指令切成两半，游戏必崩。

4. 你该怎么做：代码怎么写

4.1 调一个"参数是 double"的静态方法

x64 上 double 参数走 xmm0：

; 调 Engine.set_TimeScale(2.0)
; （set_TimeScale 没有 this，因为是静态方法）
48 B8 00 00 00 00 00 00 00 40   mov rax,0x4000000000000000   ; 2.0 的 double 位模式
66 48 0F 6E C0                  movq xmm0,rax
48 B8 <set_TimeScale 地址>       mov rax,<addr>
FF D0                           call rax
double 位模式：1.0 = 0x3FF0000000000000、2.0 = 0x4000000000000000、4.0 = 0x4010000000000000、0.5 = 0x3FE0000000000000
拿不准就用 BitConverter.DoubleToInt64Bits(2.0).ToString("X") 现算。

F2 0F 10 00                 movsd xmm0,[rax]
48 B8 <已应用地址>          mov rax,&applied
F2 0F 10 08                 movsd xmm1,[rax]
66 0F 2E C8                 ucomisd xmm1,xmm0
7A xx                       jp  跳过               ; 处理 NaN
74 xx                       je  跳过
F2 0F 11 00                 movsd [rax],xmm0       ; applied = want
48 B8 <set_TimeScale>       mov rax,<addr>
FF D0                       call rax
跳过:
这条链的前提
call 的目标必须是真方法地址（不是 0xFFFF…）。Godot 的 set_TimeScale 是 P/Invoke 到原生层的薄封装，
一般已 JIT；拿不到就按第 3 节的备选方案走"写值"。

坑 1 ★ 静态类的方法，堆里查不到（影响最大）

现象

Hook::ModifyDamage、Hook::ModifyCardRewardUpgradeOdds、Hook::ShouldTakeExtraTurn、

Hook::ModifyMerchantPrice、Hook::ModifyMaxEnergy、ScoreUtility::CalculateScore

——这些方法明明存在，methods 命令却一个都查不出来，trainer 里全显示"未JIT"。

为什么

// 我原来的取法：从"堆里该类型的第一个实例"拿到 ClrType
if (!Heap.TryGetValue(typeName, out var list) || list.Count == 0) return 结果为空;
foreach (var m in list[0].Type.Methods) { ... }

MegaCrit.Sts2.Core.Hooks.Hook 是 static class，MegaCrit.Sts2.Core.Runs.ScoreUtility 也是

⇒ 它们永远不会有实例 ⇒ Heap[typeName] 是空的 ⇒ 拿不到 ClrType ⇒ 查不到方法。

怎么改（关键三行）

ClrType TypeOf(string typeName)
{
    if (Heap.TryGetValue(typeName, out var l) && l.Count > 0) return l[0].Type;   // ① 有实例就用实例
    foreach (var m in Runtime.EnumerateModules())                                // ② 没有就查模块类型表
    {
        var t = m.GetTypeByName(typeName);
        if (t != null) return t;
    }
    return null;
}
为什么"模块类型表"能查到
EnumerateModules() 列出已加载的托管模块（这里就是 sts2.dll），
GetTypeByName() 直接读元数据类型定义表——和有没有实例无关，静态类也在里面。

自查

# 如果这条命令有输出，但你的程序查不到 ⇒ 就是踩了这个坑
dotnet run --project D:\sts2\clr\sts2clr -- <pid> allmethods MegaCrit.Sts2.Core.Hooks.Hook

真实输出（本机）：

MegaCrit.Sts2.Core.Hooks.Hook::ModifyDamage native=0x7FF8E56DE460
MegaCrit.Sts2.Core.Hooks.Hook::ModifyDamageInternal native=0x7FF8E56DEAE0

坑 2 ★ 残留补丁：强杀 trainer 后，游戏里还留着 jmp

现象

新开 trainer，除了一两个功能，其余全部"待机/安装失败"；更奇怪的是旧功能还在生效。

为什么

内存补丁不落盘，但也不会因为 trainer 退出而消失。如果 trainer 是被强杀/崩溃的，

它没走到"还原"那一步 → 方法入口仍然是 E9 <rel32>（jmp 到已经没人管的代码洞）。

现场证据（只读 dump 一个正在跑的游戏进程）：

Creature::set_Block       入口 = E9 AB E8 49 00 90 …     ← 已经被打过补丁
Creature::LoseHpInternal  入口 = E9 5B 7E 4C 00 56 …     ← 也被打过补丁

E9 就是 jmp。这时新实例读入口算"序言长度"会失败（第一条指令就是 5 字节 jmp），或者我们主动检测到 E9 就放弃。

怎么改

  1. 检测：装钩子前先看首字节，0xE9 就直接报"挂点已被占用"，不要硬写。
  2. 提示：让 UI 明确显示"挂点被占用 → 重启游戏"，而不是笼统"待机"。
  3. 恢复：正常关窗口时逐个 Restore（本项目已做）；强杀无法挽回，只能重启游戏。
var head = Read(h, patchAddr, 16);
if (head[0] == 0xE9) { 提示("挂点已被占用，请重启游戏"); return null; }
副作用：新旧补丁混在一起会"看起来能用一半"
旧补丁还在改血量，新补丁的"无限血"却没装上 —— 用户会以为"这个功能坏了"。
所以一定要把原因显示出来，别只说"待机"。

坑 3 ★ 未 JIT：.NET 是"用到才编译"

现象

新开一局，PotionRewardOdds::Roll、CalculateScore、ShouldTakeExtraTurn 全是"未JIT"。

过一会儿（打完一场 / 进奖励面板）它们就变成可用。

为什么

NativeCode == 0xFFFFFFFFFFFFFFFF 就是"这个方法还没被编译成机器码"。

奖励、结算、敌人回合调度这类方法，只有走到那个流程才会第一次 JIT。

怎么改：延迟安装 + 清缓存重试

// 主循环里每 3 秒扫一遍"还没装上的"
if (Environment.TickCount64 - lastRetry > 3000) {
    lastRetry = Environment.TickCount64;
    g.InvalidateTypes();          // ★ 关键：清掉 ClrType 缓存，否则一直读到旧的 0xFFFF
    g.Resolve();
    foreach (var f in feats) if (!f.Ready) TryInstall(f);
}
不清缓存会怎样
ClrType 会被缓存，ClrMethod.NativeCode 读到的可能还是旧值 ⇒ 明明已经 JIT 了，你的程序仍然认为"未JIT"，永远不装。

坑 4 ★ 帧钩子的"只写值"模式（刷新方法查不到时的退路）

现象

能量/星星的刷新方法 NEnergyCounter::RefreshLabel / NStarCounter::RefreshVisibility

在 ClrMD 的方法枚举里查不到（和坑 1 同类，但这次是"私有的、枚举不全"），

于是帧动作因为"CallFn 地址非法"整组作废。

怎么改：允许 CallFn = 0

// 校验放宽：只要有"要写的地方"，就允许不调刷新方法
if (Bad(a.CheckAddr) || (a.WriteAddr == 0 && Bad(a.CallFn))) { 跳过; }

// 生成机器码时：CallFn 有效才发 call，否则只写值
if (!Bad(a.CallFn) && !Bad(a.CallThis)) { ... movups 保存 xmm0 ... call rax ... }

什么时候可以只写值：

功能只写值够不够
星星✅ 够——NStarCounter._Process 每帧自己刷数字
能量⚠️ 值够打牌就行，但显示的数字要等下一次事件才更新
血量❌ 不够——NTopBarHp 纯事件驱动，必须调刷新方法（或用 setter 钩子）

坑 5 ★ decimal 返回值会让参数整体右移（ModifyDamage 的 ABI）

现象

按"普通方法"的思路去 ModifyDamage 里找 &damage，位置全错，改不到伤害。

为什么

public static decimal ModifyDamage(IRunState runState, ICombatState combatState,
                                   Creature target, Creature dealer, decimal damage, ...)

C# 的 decimal 是 16 字节结构体 ⇒ x64 上返回值要靠"调用者给的缓冲"，

约定把缓冲指针放进 rcx，于是所有参数整体右移一格：

rcx = &ret(decimal)        ← 不是 runState！
rdx = IRunState
r8  = ICombatState
r9  = target（受伤者）      ★ 门禁就靠它
栈：[rsp+0x28] = dealer
    [rsp+0x30] = &damage(decimal)   ★ 伤害值在这里
    [rsp+0x38] = props
    [rsp+0x40] = cardSource
    [rsp+0x48] = hookType
    [rsp+0x50] = previewMode
    [rsp+0x58] = out modifiers

怎么改（本项目最终用的机器码）

压 6 个寄存器后，栈参数偏移整体 +0x30 ⇒ &damage = [rsp+0x60]：

50 53 51 52 41 50 41 51        push rax,rbx,rcx,rdx,r8,r9
4C 8B 44 24 60                 mov  r8,[rsp+0x60]     ; &damage
4D 85 C0                       test r8,r8 / je 结束
48 8B 1C 24                    mov  rbx,[rsp]         ; target（刚压进去的 r9）
48 B9 <玩家Creature>            mov  rcx,玩家
48 39 CB                       cmp  rbx,rcx
74 ..                          je   玩家侧
; 敌方侧
cmp dword [flagSuper],0 / jne → mov dword [r8],9999 + 清零 4/8/0xC   ; 超级伤害
cmp dword [flagMul],0   / jne → eax=[r8]; imul eax,eax,N; 写回      ; 伤害倍率
; 玩家侧
cmp dword [flagDef],0   / jne → eax=[r8]; cdq; idiv N; 写回         ; 防御倍率
; 还原 6 个寄存器 → 补回原指令 → jmp 回
两条重载都要挂
原版样本里就是两条：ModifyDamage(IRunState,ICombatState,…) 和 (IRunState,CombatState,…)。
只挂一个，另一半调用路径不受影响 ⇒ 表现为"有时候生效有时候不生效"。

做一个能用的 trainer，"能不能挂上"比"挂得对不对"更常出问题：
先保证"每种失败都有明确原因"，再谈功能逻辑。
